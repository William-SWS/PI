entrada: imagem f (x, y) com 256 níveis de cinza
saída: imagem g(x, y) com 2 níveis de cinza
for x = 0 até M − 1 do
    for y = 0 até N − 1 do
        if f (x, y) < 128 then
            g(x, y) = 0 // cor branca
        else
            g(x, y) = 1 // cor preta
        end if
        // armazenar o erro (diferen ̧ca entre o valor exato do pixel e o valor aproximado)
        erro = f (x, y) − g(x, y) ∗ 255
        // distribuir o erro aos pixels adjacentes
        f (x + 1, y) = f (x + 1, y) + (7/16)*erro
        f (x − 1, y + 1) = f (x − 1, y + 1) + (3/16)*erro
        f (x, y + 1) = f (x, y + 1) + (5/16)*erro
        f (x + 1, y + 1) = f (x + 1, y + 1) + (1/16)*erro
    end for
end for
